// corpo asincrono caricato dal banco (c177-car3-server.mjs) all'avvio: definisce gli helper in `st`
st.phase=()=>page.evaluate(()=>window.__CPM_PHASE?window.__CPM_PHASE():null);
st.btns=()=>page.evaluate(()=>[...document.querySelectorAll('button')].filter(b=>{const r=b.getBoundingClientRect();return r.width>0&&r.height>0;}).map(b=>(b.innerText||'').trim().replace(/\n/g,' | ').slice(0,90)).filter(x=>x));
st.txt=async(n=400)=>(await testo(page)).slice(0,n).replace(/\n/g,' / ');
st.dif=(o)=>{o.ts=new Date().toISOString();fs.appendFileSync(OUT+'/difetti.jsonl',JSON.stringify(o)+'\n');return 'dif#'+(fs.readFileSync(OUT+'/difetti.jsonl','utf8').trim().split('\n').length);};
st.salva=async()=>{const d=await page.evaluate(()=>{const o={};for(let i=0;i<localStorage.length;i++){const k=localStorage.key(i);if(/^cpm/.test(k))o[k]=localStorage.getItem(k);}return o;});fs.writeFileSync(SAVEF,JSON.stringify(d));return d['cpm-v3']?d['cpm-v3'].length:0;};
st.cta=async()=>page.evaluate(()=>{const b=document.querySelector('[data-cpm="avanza23"]');if(!b)return null;const t=(b.innerText||'').replace(/\n/g,' | ');if(b.disabled)return 'DIS:'+t;b.click();return t;});
st.snap=()=>page.evaluate(()=>{try{const C=window.__CPM_CAREER;const s=C.snapshot();return s;}catch(e){return null;}});
st.riga=async(extra)=>{const s=await st.snap();if(!s)return null;
 const r={s:s.season,w:s.week,age:s.age,club:s.club&&s.club.n,lega:s.club&&s.club.lg,naz:!!(s.nationalCaps),caps:s.nationalCaps||0,natGoals:s.nationalGoals,ovr:s.ovr,gol:s.goals,ass:s.assists,pres:s.matches,totGol:s.totalGoals,form:s.form,morale:s.morale,fatica:s.fatigue,banca:s.bankBalance,pro:s.proStatus,inj:s.injury?s.injury.type+':'+s.injury.weeksRemaining:null,...(extra||{})};
 fs.appendFileSync(OUT+'/settimane.jsonl',JSON.stringify(r)+'\n');return r;};
const NAV=/^(Salva|Calendario|La tua settimana|Obiettivi|Ultime notizie|☕|💡|HOME|STAGIONE|CLUB|CARRIERA|AGENTE|UFFICIO|OPZIONI)/;
st.rot=0;
/* una passata di "sbroglio": chiude/risolve la finestra che copre la home. Ritorna descrizione o null */
st.sbroglia=async()=>{const b=await st.btns();const ph=await st.phase();
 const t=await testo(page);
 for(const r of [/^Gioca la partita/]){}
 const pri=[/^Tieni #/,/^Capito/,/^Avanti/,/^Continua/,/^Conferma/,/^Prosegui/,/^Ok\b/i,/^🏟️ Torna alla dashboard/,/^Torna alla dashboard/,/^🎉/,/^Chiudi$/,/^Chiudi —/];
 for(const r of pri){const x=b.find(y=>r.test(y));if(x){await tocca(page,x.split(' | ')[0],2000);return 'click:'+x.slice(0,50);}}
 return null;};
/* in match: attende il fischio finale (tabellino a schermo) con autoplay seedato */
st.inMatch=async(seed,maxS=240)=>{await page.evaluate((s)=>{try{window.__CPM_AUTOPLAY(true,{seed:s,policy:'seeded',tickMs:300});}catch(e){}},seed);
 st.hl=[];let last='';const t0=Date.now();
 while((Date.now()-t0)<maxS*1000){await sleep(1100);const b=await st.btns();
  if(b.some(x=>/Torna alla dashboard|Risultati provino|^Continua|Chiudi$/.test(x))&&!b.some(x=>/^⏸/.test(x))){st.fineTxt=await testo(page);return 'fine';}
  const ph=await page.evaluate(()=>window.__CPM_PHASE?window.__CPM_PHASE():null);
  if(ph==='hl_result'&&last!=='hl_result'){const info=await page.evaluate(()=>{try{const tl=(window.__CPM_TIMELINE?window.__CPM_TIMELINE():[]).slice(-3);const sc=window.__CPM_SCORE?window.__CPM_SCORE():null;const stt=window.__CPM_STATE?window.__CPM_STATE():null;return {tl,sc,ball:stt&&stt.ball,txt:document.body.innerText.slice(0,500).replace(/\n/g,' / ')};}catch(e){return {err:String(e)};}});st.hl.push(info);}
  last=ph;}
 return 'timeout';};
st.scelta=async()=>{const b=await st.btns();const cand=b.filter(x=>!NAV.test(x)&&!/^Salta|^Non ora|^Chiudi|^⏸|^2×|^⚠️|^1×|^1,5×/.test(x));
 if(cand.length<1)return null;const ok=cand.filter(x=>!/🔒/.test(x));const arr=ok.length?ok:cand;const x=arr[(st.rot++)%arr.length];await tocca(page,x.split(' | ')[0].replace(/🔒.*$/,'').trim(),2500);return 'scelta:'+x.slice(0,70);};
/* una settimana intera. opts: live (bool), seed. Ritorna riepilogo. */
st.sett=async(opts={})=>{const s0=await st.snap();const k0=s0.season*100+s0.week;const ev=[];let live=!!opts.live;
 for(let i=0;i<70;i++){
  const s=await st.snap();if(s&&(s.season*100+s.week)!==k0&&i>0){ev.push('→sett '+s.season+'/'+s.week);break;}
  const b=await st.btns();const t=await st.txt(250);
  if(!opts.chiudi&&b.some(x=>/Chiudi la stagione/.test(x))){ev.push('STOP-FINE-STAGIONE');break;}
  if(b.some(x=>/^Gioca la partita/.test(x))){const tt=await testo(page);const nat=/Nazionale|Europe|Mondiale|Qualific|Amichevole Internaz|Coppa delle Nazioni|Convocaz/i.test(tt.slice(0,400));if(nat)ev.push('NAT-PROMPT:'+tt.slice(0,160).replace(/\n/g,' / '));const q=(live||nat)?'Gioca la partita':'Simula e avanza';await tocca(page,q,3000);ev.push('prompt:'+q);await sleep(1500);continue;}
  if(b.some(x=>/^Salta e gioca/.test(x))){await tocca(page,opts.vedi?'Vedi le formazioni':'Salta e gioca',3000);ev.push('prematch');await sleep(1500);continue;}
  if(b.some(x=>/^Ingresso in campo/.test(x))){await tocca(page,opts.ingresso?'Ingresso in campo':"Salta l'ingresso",3000);ev.push('ingresso');await sleep(2000);continue;}
  if(b.some(x=>/^⏸/.test(x))){const r=await st.inMatch(opts.seed||7);ev.push('match:'+r);fs.appendFileSync(OUT+'/partite-live.jsonl',JSON.stringify({k:k0,hl:st.hl,fine:(st.fineTxt||'').slice(0,2500)})+'\n');continue;}
  if(b.some(x=>/Torna alla dashboard/.test(x))){opts.postFoto&&await foto(page,opts.postFoto);ev.push('postmatch:'+(await st.txt(120)));await tocca(page,'Torna alla dashboard',3000);await sleep(1500);continue;}
  const sb=await st.sbroglia();if(sb){ev.push(sb);await sleep(900);continue;}
  const hasCta=b.some(x=>/^(🏁|⚽|⚡|▶|➡|📅|🏆|⏭|✅)/.test(x)&&!NAV.test(x));
  const extra=b.filter(x=>!NAV.test(x)&&!/^(⚡|🏁|⚽|▶|➡|📅|🏆|⏭|✅|🏥)/.test(x));
  const mod=extra.length>0||/Impulso|SETTIMANA \d+ ·|scelta|Scegli|decid/i.test(t);
  if(mod){const sc=await st.scelta();if(sc){ev.push(sc);await sleep(1200);continue;}}
  const c=await st.cta();ev.push('cta:'+c);if(c===null||/^DIS/.test(c)){const sc=await st.scelta();ev.push('?'+sc+'|'+t.slice(0,100));if(!sc){await sleep(1500);}}
  await sleep(1400);
 }
 return ev;};

st.corri=async(n,o={})=>{const out=[];const e0=st.errori.length;for(let k=0;k<n;k++){const s=await st.snap();if(!s){out.push('NOSNAP');break;}
  const scr=await page.evaluate(()=>window.__CPM_CAREER.screen());
  if(scr&&/seasonEnd|seasonAwards|careerEnd|proTransition/.test(scr)){out.push('SCHERMATA:'+scr);break;}
  if((await st.btns()).some(x=>/Chiudi la stagione/.test(x))){out.push('FINE-STAGIONE-PRONTA');break;}
  st.cont=(st.cont||0)+1;const live=(st.cont%(o.every||4))===0;
  const ev=await st.sett({live,seed:(st.cont*37)%1000+3,vedi:false,ingresso:live&&(st.cont%8===0)});
  const r=await st.riga({live,ev:ev.filter(x=>!/^click|^cta|^→/.test(x)).map(x=>x.slice(0,140))});out.push((r?r.s+'/'+r.w+' ovr'+r.ovr+' g'+r.gol:'?')+(live?' LIVE':'')+(ev.some(x=>/NAT-PROMPT/.test(x))?' NAT':''));
  if(ev.some(x=>/timeout|\?null/.test(x)))out.push('ANOMALIA:'+JSON.stringify(ev).slice(0,300));
  if(st.errori.length>e0+30)break;}
 return {out,errNuovi:st.errori.slice(e0).slice(0,5),nErr:st.errori.length-e0};};
